cmake_minimum_required(VERSION 3.21)
project(BlackboxDesktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Qml DBus Quick)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(LIBVTERM QUIET IMPORTED_TARGET vterm>=0.3.3)
endif()
if(TARGET PkgConfig::LIBVTERM)
    set(terminal_emulator PkgConfig::LIBVTERM)
else()
    file(GLOB libvterm_sources CONFIGURE_DEPENDS vendor/libvterm/src/*.c)
    add_library(blackbox_vterm STATIC ${libvterm_sources})
    target_include_directories(blackbox_vterm PUBLIC vendor/libvterm/include)
    set_target_properties(blackbox_vterm PROPERTIES POSITION_INDEPENDENT_CODE ON AUTOMOC OFF C_STANDARD 99 C_VISIBILITY_PRESET hidden)
    set(terminal_emulator blackbox_vterm)
endif()
add_library(blackboxdesktop MODULE bridge.cpp terminal.cpp terminal.h)
target_link_libraries(blackboxdesktop PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::DBus Qt6::Quick ${terminal_emulator} util)
set_target_properties(blackboxdesktop PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/native")
configure_file(qmldir "${CMAKE_BINARY_DIR}/native/qmldir" COPYONLY)
